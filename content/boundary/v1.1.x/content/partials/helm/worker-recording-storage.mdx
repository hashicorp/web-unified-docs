If the worker stores session recordings, enable recording persistence and set `recording_storage_path` in `worker.config` to the same path.

<Note>

To create Remote Desktop Protocol (RDP) session recording exports, the worker environment requires FFmpeg 5.1.0 or later and `libsvtav1`. Refer to [Configure worker storage](/boundary/docs/session-recording/configuration/configure-worker-storage#ffmpeg) for requirements.

</Note>

Save the following complete HCL and values as `worker-values.yaml`:

```yaml
worker:
  config: |
    disable_mlock = true

    listener "tcp" {
      address = "0.0.0.0:9202"
      purpose = "proxy"
    }

    listener "tcp" {
      address       = "0.0.0.0:9203"
      purpose       = "ops"
      tls_disable   = "{{ .Values.tls.ops.disabled }}"
      tls_cert_file = "{{ .Values.tls.mountPath }}/tls.crt"
      tls_key_file  = "{{ .Values.tls.mountPath }}/tls.key"
    }

    worker {
      name                   = "boundary-worker"
      initial_upstreams      = ["boundary-controller-cluster:9201"]
      recording_storage_path = "/boundary/recording"

      tags {
        type = ["worker", "recording"]
      }
    }

    kms "aead" {
      purpose   = "worker-auth"
      aead_type = "aes-gcm"
      key       = "<base64-encoded-32-byte-key>"
      key_id    = "<worker-auth-key-id>"
    }

    events {
      audit_enabled        = true
      sysevents_enabled    = true
      observations_enabled = true
      sink "stderr" {
        name        = "all-events"
        event_types = ["*"]
        format      = "cloudevents-json"
      }
    }
  persistence:
    recording:
      enabled: true
      size: 10Gi
      path: /boundary/recording
```

Install the worker Helm chart:

```shell-session
$ helm install boundary-worker hashicorp/boundary-worker \
  --namespace boundary \
  --values worker-values.yaml \
  --wait
```

Ensure the recording storage path in `worker.config` matches `worker.persistence.recording.path`.
