The typical workflow for controller-led registration is as follows. For more details about controller-led registration, refer to [Controller-led registration](/boundary/docs/workers/registration/controller-led):

1. Create the worker resource in Boundary or HCP Boundary.
2. Obtain the controller-generated activation token.
3. Supply the token using one of the methods below.
4. Install the worker Helm chart.
5. Confirm the worker registers and persists its credentials to auth storage.

The worker uses the activation token during initial bootstrap. After successful registration, the worker uses persisted credentials from auth storage.

The chart supports three ways to supply the activation token.

<Note>

The plaintext and `extraEnv` methods store the token in the Helm release Secret, which means it is visible to anyone who can run `helm get values boundary-worker --namespace boundary`. Use the `secretRefs` method if you need the token to stay out of Helm release storage entirely.

</Note>

### Plaintext token

Embed the token directly in `worker.config`. Suitable for development only. The token is visible to anyone who can read the ConfigMap.

Save the following complete HCL and values as `worker-values.yaml`:

```yaml
worker:
  config: |
    disable_mlock = true

    listener "tcp" {
      address = "0.0.0.0:9202"
      purpose = "proxy"
    }

    listener "tcp" {
      address       = "0.0.0.0:9203"
      purpose       = "ops"
      tls_disable   = "{{ .Values.tls.ops.disabled }}"
      tls_cert_file = "{{ .Values.tls.mountPath }}/tls.crt"
      tls_key_file  = "{{ .Values.tls.mountPath }}/tls.key"
    }

    worker {
      name                                  = "boundary-worker"
      initial_upstreams                     = ["boundary-controller-cluster:9201"]
      auth_storage_path                     = "/var/lib/boundary"
      controller_generated_activation_token = "<activation-token>"

      tags {
        type = ["worker", "controller-led"]
      }
    }

    events {
      audit_enabled        = true
      sysevents_enabled    = true
      observations_enabled = true
      sink "stderr" {
        name        = "all-events"
        event_types = ["*"]
        format      = "cloudevents-json"
      }
    }
  persistence:
    authStorage:
      enabled: true
      size: 1Gi
      path: /var/lib/boundary
```

Install the worker Helm chart:

```shell-session
$ helm install boundary-worker hashicorp/boundary-worker \
  --namespace boundary \
  --values worker-values.yaml \
  --wait
```

### Token via extraEnv

Set `controller_generated_activation_token` to the `env://` reference in `worker.config` and supply the token as a key-value pair in `extraEnv`. Save the following complete HCL and values as `worker-values.yaml`:

```yaml
extraEnv:
  - name: BOUNDARY_WORKER_CONTROLLER_GENERATED_ACTIVATION_TOKEN
    value: "<activation-token>"

worker:
  config: |
    disable_mlock = true

    listener "tcp" {
      address = "0.0.0.0:9202"
      purpose = "proxy"
    }

    listener "tcp" {
      address       = "0.0.0.0:9203"
      purpose       = "ops"
      tls_disable   = "{{ .Values.tls.ops.disabled }}"
      tls_cert_file = "{{ .Values.tls.mountPath }}/tls.crt"
      tls_key_file  = "{{ .Values.tls.mountPath }}/tls.key"
    }

    worker {
      name                                  = "boundary-worker"
      initial_upstreams                     = ["boundary-controller-cluster:9201"]
      auth_storage_path                     = "/var/lib/boundary"
      controller_generated_activation_token = "env://BOUNDARY_WORKER_CONTROLLER_GENERATED_ACTIVATION_TOKEN"

      tags {
        type = ["worker", "controller-led"]
      }
    }

    events {
      audit_enabled        = true
      sysevents_enabled    = true
      observations_enabled = true
      sink "stderr" {
        name        = "all-events"
        event_types = ["*"]
        format      = "cloudevents-json"
      }
    }
  persistence:
    authStorage:
      enabled: true
      size: 1Gi
      path: /var/lib/boundary
```

Install the worker Helm chart:

```shell-session
$ helm install boundary-worker hashicorp/boundary-worker \
  --namespace boundary \
  --values worker-values.yaml \
  --wait
```
