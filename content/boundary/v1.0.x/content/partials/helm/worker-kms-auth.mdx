Boundary only supports KMS-backed worker authentication in self-managed deployments. For more information, refer to [External KMS registration](/boundary/docs/workers/registration/kms). It is not available for HCP Boundary workers.

For self-managed deployments, configure the worker HCL with an external KMS worker-auth stanza that matches the controllers. The following example uses AWS KMS. Configure workload credentials that grant the pod access to the KMS key, and replace the placeholders for your environment. Save the complete HCL and values as `worker-values.yaml`:

```yaml
worker:
  config: |
    disable_mlock = true

    listener "tcp" {
      address = "0.0.0.0:9202"
      purpose = "proxy"
    }

    listener "tcp" {
      address       = "0.0.0.0:9203"
      purpose       = "ops"
      tls_disable   = "{{ .Values.tls.ops.disabled }}"
      tls_cert_file = "{{ .Values.tls.mountPath }}/tls.crt"
      tls_key_file  = "{{ .Values.tls.mountPath }}/tls.key"
    }

    worker {
      name              = "boundary-worker"
      initial_upstreams = ["boundary-controller-cluster:9201"]

      tags {
        type = ["worker", "kms"]
      }
    }

    kms "awskms" {
      purpose    = "worker-auth"
      region     = "<aws-region>"
      kms_key_id = "<aws-kms-key-id>"
    }

    events {
      audit_enabled        = true
      sysevents_enabled    = true
      observations_enabled = true
      sink "stderr" {
        name        = "all-events"
        event_types = ["*"]
        format      = "cloudevents-json"
      }
    }
  persistence:
    authStorage:
      enabled: false
      path: /var/lib/boundary
```

Install the worker Helm chart:

```shell-session
$ helm install boundary-worker hashicorp/boundary-worker \
  --namespace boundary \
  --values worker-values.yaml \
  --wait
```
